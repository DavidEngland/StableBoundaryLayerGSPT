```julia
# scripts/generate_symbols.jl
using YAML

function generate_latex_table(spec::Dict)
    io = IOBuffer()
    println(io, "\\begin{table}[htbp]")
    println(io, "\\centering")
    println(io, "\\caption{List of symbols, mathematical definitions, physical units, and model variable bindings.}")
    println(io, "\\label{tab:list_of_symbols}")
    println(io, "\\small")
    println(io, "\\begin{tabular}{llll}")
    println(io, "\\toprule")
    println(io, "Symbol & Description & Units & Code Mapping \\\\")
    println(io, "\\midrule")

    for cat in spec["categories"]
        cat_symbols = filter(s -> s["category"] == cat["id"], spec["symbols"])
        if !isempty(cat_symbols)
            println(io, "\\multicolumn{4}{l}{\\textbf{$(cat["name"])}} \\\\")
            println(io, "\\midrule")
            for s in cat_symbols
                sym = "\\(\$(s["symbol_tex"])\\)"
                desc = s["description"]
                units = s["units_tex"] == "-" ? "---" : "\\(\$(s["units_tex"])\\)"
                code = "\\texttt{\$(s["code_key"])}"
                println(io, "\$sym & \$desc & \$units & \$code \\\\")
            end
            println(io, "\\midrule")
        end
    end

    println(io, "\\bottomrule")
    println(io, "\\end{tabular}")
    println(io, "\\end{table}")
    return String(take!(io))
end

```

In Equation \eqref{eq:heff_emergent}, **$c_1$ and $c_2$ are dimensionless empirical scaling constants** that calibrate the two distinct physical constraints acting on the stable boundary layer (SBL) depth:

* **$c_1$ (Mechanical / Rotational Limit Constant):** Scales the neutral **Rossby–Montgomery (Zilitinkevich)** boundary layer height limit, $h \sim \frac{u_*}{\vert{}f\vert{}}$. It governs boundary layer depth in shear-dominated or weakly stable conditions, where planetary rotation ($\vert{}f\vert{}$) and surface mechanical friction velocity ($u_*$) establish the vertical depth scale.
* **$c_2$ (Buoyancy / Stratification Limit Constant):** Scales the **stratification-limited (Deardorff)** height limit, $h \sim \frac{\sqrt{e}}{N}$. It governs boundary layer depth in strongly stable nocturnal conditions, where buoyant suppression truncates vertical turbulent eddies against the static thermal stratification (quantified by the Brunt–Väisälä frequency $N$).

### Physical Function in the Minimum Operator

By combining these two constants within the $\min(\dots)$ envelope:

1. Under **weak stability (high winds / strong shear)**, the $c_1$ term dominates, preventing unphysically deep boundary layers by capping $h_{\text{eff}}$ at the rotational Ekman scale.
2. Under **strong stability (calm winds / clear skies)**, the $c_2$ term takes over, collapsing $h_{\text{eff}}$ proportional to the turbulent velocity scale $\sqrt{e}$ relative to $N$. This dynamic truncation prevents the artificial runaway surface cooling typically seen in local closures when turbulence decays.

```julia
using YAML, Glob

# 1. Load canonical TeX symbols from SSOT
symbols_spec = YAML.load_file("spec/symbols.yaml")
known_tex = Set(String[s["symbol_tex"] for s in symbols_spec["symbols"]])

# Regex matching math macros: \mathrm{...}, \mathcal{...}, subscripted patterns
math_regex = r"\\(?:mathrm|mathcal)\{[A-Za-z0-9_]+\}|[A-Za-z]_[A-Za-z0-9]+"

for file in template_files

        end
    end
end

println("--- Unmapped Math Candidates ---")
```

### Updated Code (`scripts/assemble_manuscript.jl`)

Below is the production-grade implementation of the figure metadata resolver and LaTeX figure generator for `assemble_manuscript.jl`:

```julia
#!/usr/bin/env julia
# scripts/assemble_manuscript.jl - Dynamic Figure Caption & Spec Integration

using JSON

const DEFAULT_FIGURE_SPEC_PATH = "spec/figure_metadata.json"

"""
    load_figure_metadata(path::String = DEFAULT_FIGURE_SPEC_PATH)

Loads figure titles, SCM captions, labels, and acronym rules from the SSOT JSON spec.
"""
function load_figure_metadata(path::String = DEFAULT_FIGURE_SPEC_PATH)
    isfile(path) || error("Figure metadata JSON not found at $(path)")
    return JSON.parsefile(path)
end

"""
    get_figure_info(meta::Dict, fig_key::String)

Resolves caption/title and label for a given figure key or file stem.
Supports exact matches, alias stems, and SCM figure IDs (e.g., 'fig01').
"""
function get_figure_info(meta::Dict{String, Any}, fig_key::String)
    # Strip extension if present (e.g., "figure_fold_illusion.pdf" -> "figure_fold_illusion")
    stem = replace(fig_key, r"\.(pdf|png|svg|eps)$"i => "")

    figures_dict = get(meta, "figures", Dict{String, Any}())
    scm_dict = get(meta, "scm_figures", Dict{String, Any}())
    acronyms = get(meta, "acronyms", Dict{String, Any}())

    # 1. Direct match in `figures`
    if haskey(figures_dict, stem)
        entry = figures_dict[stem]
        return (
            caption = String(entry["title"]),
            label   = String(entry["label"])
        )
    end

    # 2. Direct match in `scm_figures` (e.g., "fig01")
    if haskey(scm_dict, stem)
        entry = scm_dict[stem]
        return (
            caption = String(entry["caption"]),
            label   = String(entry["label"])
        )
    end

    # 3. Normalized alias lookup (handling "fig_" vs "figure_" prefixes)
    alt_stem = startswith(stem, "fig_") ? replace(stem, "fig_" => "figure_") :
               startswith(stem, "figure_") ? replace(stem, "figure_" => "fig_") : stem

    if haskey(figures_dict, alt_stem)
        entry = figures_dict[alt_stem]
        return (
            caption = String(entry["title"]),
            label   = String(entry["label"])
        )
    end

    # 4. Graceful fallback: derive title from stem using acronym map
    words = split(replace(stem, "_" => " "), " ")
    formatted_words = map(w -> get(acronyms, lowercase(w), titlecase(w)), words)
    fallback_title = join(formatted_words, " ")
    fallback_label = "fig:$(stem)"

    @warn "Figure key '$(fig_key)' not found in $(DEFAULT_FIGURE_SPEC_PATH). Using derived fallback."
    return (caption = fallback_title, label = fallback_label)
end

"""
    render_figure_environment(fig_key::String, image_rel_path::String, meta::Dict;
                              width::String = "0.95\\textwidth",
                              position::String = "htbp")

Renders a standard, publication-ready LaTeX figure block.
"""
function render_figure_environment(fig_key::String, image_rel_path::String, meta::Dict;
                                  width::String = "0.95\\textwidth",
                                  position::String = "htbp")
    info = get_figure_info(meta, fig_key)

    lines = String[
        "\\begin{figure}[$(position)]",
        "    \\centering",
        "    \\includegraphics[width=$(width)]{$(image_rel_path)}",
        "    \\caption{$(info.caption)}",
        "    \\label{$(info.label)}",
        "\\end{figure}"
    ]
    return join(lines, "\n")
end

"""
    inject_figure_metadata!(context::Dict{String, String}, meta::Dict)

Populates Mustache template context with captions and labels for mustache-driven LaTeX builds.
"""
function inject_figure_metadata!(context::Dict{String, String}, meta::Dict)
    # Populate general figures
    for (key, val) in get(meta, "figures", Dict())
        context["fig_title_$(key)"] = String(val["title"])
        context["fig_label_$(key)"] = String(val["label"])
    end

    # Populate SCM figures
    for (key, val) in get(meta, "scm_figures", Dict())
        context["fig_caption_$(key)"] = String(val["caption"])
        context["fig_label_$(key)"] = String(val["label"])
    end

    return context
end

```
